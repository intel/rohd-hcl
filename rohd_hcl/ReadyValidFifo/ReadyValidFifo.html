@Deprecated(
  'Use ReadyValidQueue with an explicit queue policy and storage.',
)
ReadyValidFifo({
  required Logic clk,
  required Logic reset,

  /// Templates describing the shape of the upstream/downstream data
  required ReadyValidInterface<LogicType> upstream,
  required ReadyValidInterface<LogicType> downstream,
  this.depth = 8,
  this.allowSameCycleReplacement = false,
  String name = 'ready_valid_fifo',
}) : super(
        definitionName: allowSameCycleReplacement
            ? 'ReadyValidFifo_R1_${upstream.data.name}'
            : 'ReadyValidFifo_${upstream.data.name}',
        name: name,
      ) {
  // Build the internal logic
  final writeEnable = Logic(name: '${name}_write_en');
  final readEnable = Logic(name: '${name}_read_en');

  // Add clock/reset ports and capture the module-local signals
  this.clk = addInput('clk', clk);
  this.reset = addInput('reset', reset);

  this.upstream = addPairInterfacePorts(
    upstream,
    PairRole.consumer,
    uniquify: (original) => 'upstream_$original',
  );
  this.downstream = addPairInterfacePorts(
    downstream,
    PairRole.provider,
    uniquify: (original) => 'downstream_$original',
  );

  // Use the upstream data shape as a template for FIFO element.
  final writeData = this.upstream.data.cloneTyped();
  final binding = FifoCoreFunctor<LogicType>(
    depth: depth,
    generateOccupancy: true,
  ).construct(
    FifoCoreRequest<LogicType>(
      clk: this.clk,
      reset: this.reset,
      enqueue: writeEnable,
      dequeue: readEnable,
      payload: writeData,
      requirements: FifoCoreRequirements(
        fullReplacement: allowSameCycleReplacement,
      ),
    ),
    name: '${name}_fifo',
  );
  fifo = binding.module;

  ReadyValidIngressAttachmentPolicy<LogicType>(
    supportsSameCycleReplacement: allowSameCycleReplacement,
    storageDepth: depth,
  ).attach(
    upstream: this.upstream,
    canAccept: allowSameCycleReplacement ? binding.canEnqueue : ~binding.full,
    accept: writeEnable,
    payload: writeData,
  );
  ReadyValidEgressAttachmentPolicy<LogicType>().attach(
    downstream: this.downstream,
    offerValid: binding.outputValid,
    payload: binding.output,
    transfer: readEnable,
  );
}
